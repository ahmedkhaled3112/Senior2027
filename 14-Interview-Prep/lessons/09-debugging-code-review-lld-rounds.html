export type LineItem = { sku: string; unitPrice: number; qty: number };
export type Cart = { tenantId: string; items: LineItem[] };
export type Invoice = { id: string; tenantId: string; total: number };

export const VAT_RATE = 0.14;

// الإجمالي بالـ VAT بالجنيه
export function invoiceTotal(cart: Cart): number {
  const subtotal = cart.items.reduce((sum, item) => sum + item.unitPrice * item.qty, 0);
  return Math.round((subtotal + subtotal * VAT_RATE) * 100) / 100;
}

// معاينة: الـ cart هتبقى بكام لو طبّقنا الـ promo ده؟
export function previewPromo(cart: Cart, percentOff: number): number {
  for (const item of cart.items) item.unitPrice *= 1 - percentOff / 100;
  return invoiceTotal(cart);
}

// رقم الصفحة بيبدأ من 1
export function pageOfInvoices(rows: Invoice[], page: number, pageSize: number): Invoice[] {
  const start = (page - 1) * pageSize;
  return rows.slice(start, start + pageSize - 1);
}
English

Model solution. Work one loop per symptom: reproduce, isolate, fix, verify. Don't read the file hunting for something that looks wrong. Open with a plan you can say out loud: "Three symptoms, probably three causes. I'll take the invoice total first, because a wrong amount is the most expensive one to ship." A workable split of the 30 minutes is 3 to read and ask, 5 to write three failing tests, 15 for three loops, 5 for neighbouring cases and 2 to summarise.

Reproduce. Write one failing test per symptom before touching the module. Three red tests prove the reports are real, prove your harness runs, and define done.

Isolate by shrinking the input, with one hypothesis at a time. Say each hypothesis before you run anything, in a form that can be proved wrong. For the total: cut three lines to one, then one line to a number, and say "I think the sum is fine and the rounding step sees a value just below .5". The output confirms it: 9.405 has no exact binary form and is stored as 9.40499…, so times 100 it is 940.4999999999999 and Math.round goes down. For the cart: "I think something writes to the cart". Deep-freeze the fixture and the first write throws with the exact line. For the page: one row short is a difference of exactly one, and slice takes an exclusive end, so the - 1 is wrong.

Fix the cause, then verify. Move money to integer minor units and the rate to basis points, so the multiplication is exact and only one division happens at the end. Make the promo preview a pure function that returns a new cart. Drop the - 1. Then rerun the three tests and add at least one neighbouring case each: every subtotal up to 2,000.00 whose VAT lands exactly on .5, an empty cart, a frozen cart, the last partial page, a page past the end, and page 0. "Should work now" is not verification. Money storage is covered in section 07, Money.

// الحل: الفلوس بالقروش (integers) والـ VAT بالـ basis points
export type LineItem = { sku: string; unitPriceMinor: number; qty: number };
export type Cart = { tenantId: string; items: LineItem[] };
export type Invoice = { id: string; tenantId: string; totalMinor: number };

export const VAT_BPS = 1_400;

function assertMinor(value: number): void {
  if (!Number.isSafeInteger(value) || value < 0) throw new RangeError(`bad minor amount: ${value}`);
}

export function invoiceTotalMinor(cart: Cart): number {
  let subtotal = 0;
  for (const item of cart.items) {
    assertMinor(item.unitPriceMinor);
    subtotal += item.unitPriceMinor * item.qty;
  }
  assertMinor(subtotal);
  // الحساب ده integer الأول وبعدين division، فـ x.5 بتتمثّل بالظبط
  return subtotal + Math.round((subtotal * VAT_BPS) / 10_000);
}

// الحل: الدالة نقية، بترجع cart جديدة وعمرها ما بتلمس اللي جالها
export function withPromo(cart: Cart, percentOff: number): Cart {
  if (percentOff < 0 || percentOff > 100) throw new RangeError(`bad percent: ${percentOff}`);
  return {
    ...cart,
    items: cart.items.map((item) => ({
      ...item,
      unitPriceMinor: Math.round((item.unitPriceMinor * (100 - percentOff)) / 100),
    })),
  };
}

export const previewPromoMinor = (cart: Cart, percentOff: number): number =>
  invoiceTotalMinor(withPromo(cart, percentOff));

// الحل: slice بياخد end غير شامل، فمفيش ⁦-1⁩
export function pageOfInvoices(rows: Invoice[], page: number, pageSize: number): Invoice[] {
  if (!Number.isInteger(page) || page < 1) throw new RangeError(`page must be >= 1, got ${page}`);
  if (!Number.isInteger(pageSize) || pageSize < 1) throw new RangeError(`bad page size: ${pageSize}`);
  const start = (page - 1) * pageSize;
  return rows.slice(start, start + pageSize);
}

الحل النموذجي. اشتغل دايرة لكل عَرَض: reproduce ثم isolate ثم fix ثم verify. متقعدش تقرا الملف بتدوّر على حاجة شكلها غلط. ابدأ بخطة تقولها بصوت عالي: "عندي 3 أعراض، غالبًا 3 أسباب. هبدأ بإجمالي الفاتورة لأن الرقم الغلط أغلى حاجة توصل للعميل". تقسيم الـ 30 دقيقة اللي بينفع: 3 للقراية والأسئلة، 5 لكتابة 3 tests فاشلين، 15 للتلات دواير، 5 للحالات الجارة، و2 للتلخيص.

Reproduce. اكتب test فاشل لكل عَرَض قبل ما تلمس الـ module. التلات tests الحمر بيثبتوا إن التقارير حقيقية، وإن الـ harness شغال، وبيحددوا يعني إيه "خلصت".

Isolate بتصغير الـ input و hypothesis واحد في المرة. قول كل hypothesis قبل ما تشغّل أي حاجة، وبصيغة ينفع تتثبت إنها غلط. للإجمالي: صغّر التلات سطور لسطر، والسطر لرقم، وقول "أعتقد إن الجمع سليم وخطوة التقريب شايفة رقم تحت .5 بشوية". الـ output بيأكد: 9.405 ملهاش تمثيل binary مظبوط فبتتخزن 9.40499…، فلما تتضرب في 100 تبقى 940.4999999999999 وMath.round بيقرّب لتحت. للـ cart: "أعتقد إن فيه حاجة بتكتب في الـ cart". اعمل deep freeze للـ fixture وأول كتابة هترمي error بالسطر بالظبط. للصفحة: ناقص صف واحد معناها فرق 1 بالظبط، وslice الـ end بتاعه غير شامل، يبقى الـ - 1 غلط.

صلّح السبب وبعدين verify. حوّل الفلوس لأرقام صحيحة بالقروش والنسبة لـ basis points، عشان الضرب يبقى مظبوط وتبقى فيه قسمة واحدة في الآخر. خلي معاينة الـ promo pure function بترجع cart جديدة. واشيل الـ - 1. بعدها شغّل التلات tests تاني وزوّد حالة جارة على الأقل لكل واحد: كل subtotal لحد 2,000.00 الـ VAT بتاعه بييجي على .5 بالظبط، cart فاضية، cart متجمدة، آخر صفحة ناقصة، صفحة بعد الآخر، وصفحة صفر. Trade-off: integers, BigInt or a decimal library A JavaScript number holds integers exactly up to 253, which is about 90 trillion pounds in piastres, so integer minor units are enough for most invoices and they stay fast and JSON-friendly. Use BigInt or a decimal library when amounts can exceed that or when rates need more digits than basis points give you. In Postgres, integer minor units or numeric both work; never float or double precision for money.
الـ number في JavaScript بيمسك integers بدقة لحد 253، وده حوالي 90 تريليون جنيه بالقروش، فالقروش الـ integers كفاية لمعظم الفواتير وسريعة وسهلة في الـ JSON. استخدم BigInt أو مكتبة decimal لو المبالغ ممكن تعدي الحد ده أو لو النسب محتاجة أرقام أكتر من اللي الـ basis points بتديها. في Postgres الـ minor units الـ integer أو numeric الاتنين ينفعوا؛ بس عمرك ما تستخدم float أو double precision للفلوس.

import { setTimeout as sleep } from "node:timers/promises";

// بديل صغير لـ DB: كل عملية بتاخد من 0 لـ 3 ms زي شبكة حقيقية
export class WalletStore {
  #balances = new Map<string, number>();
  #random: () => number;

  constructor(random: () => number = Math.random) {
    this.#random = random;
  }

  seed(walletId: string, balanceMinor: number): void {
    this.#balances.set(walletId, balanceMinor);
  }

  peek(walletId: string): number {
    return this.#balances.get(walletId) ?? 0;
  }

  async read(walletId: string): Promise<number> {
    await sleep(Math.floor(this.#random() * 4));
    return this.peek(walletId);
  }

  async write(walletId: string, balanceMinor: number): Promise<void> {
    await sleep(Math.floor(this.#random() * 4));
    this.#balances.set(walletId, balanceMinor);
  }

  // الحل: الفحص والخصم في خطوة واحدة جوه الـ store، زي UPDATE ... WHERE balance >= $1
  async debitIfEnough(walletId: string, amountMinor: number): Promise<number | null> {
    await sleep(Math.floor(this.#random() * 4));
    const balance = this.peek(walletId);
    if (balance < amountMinor) return null;
    this.#balances.set(walletId, balance - amountMinor);
    return balance - amountMinor;
  }
}

  walletId: string,
  amountMinor: number,

): Promise<void> {

  trace(`read ${balance}`);

  await store.write(walletId, balance - amountMinor);

}

type Gateway = { refund: (paymentRef: string, amountCents: number) => Promise<void> };
type TenantRequest = FastifyRequest & { tenantId: string };
type OrderRow = { id: string; payment_ref: string; total_cents: number };
type RefundBody = { orderId: string; amountCents: number; reason: string };

function parseBody(raw: unknown): RefundBody | null {
  const b = raw as Partial<RefundBody> | null;
  if (typeof b?.orderId !== "string" || typeof b.reason !== "string") return null;
  if (!Number.isSafeInteger(b.amountCents) || (b.amountCents ?? 0) <= 0) return null;
  return b as RefundBody;
}

export function refundRoutes(app: FastifyInstance, db: Db, gateway: Gateway) {
  app.post("/refunds", async (req, reply) => {
    const { tenantId } = req as TenantRequest;
    const body = parseBody(req.body);
    if (!body) return reply.code(400).send({ error: "invalid body" });
    const key = req.headers["idempotency-key"];
    if (typeof key !== "string" || key === "") {
      return reply.code(400).send({ error: "Idempotency-Key required" });
    }

    // (1) الـ id بقى parameter والـ tenant جوه الـ WHERE: الاتنين في سطر واحد
    const orders = await db.query<OrderRow>(
      "SELECT id, payment_ref, total_cents FROM orders WHERE id = $1 AND tenant_id = $2",
      [body.orderId, tenantId],
    );
    const order = orders.rows[0];
    if (!order) return reply.code(404).send({ error: "order not found" });
    if (body.amountCents > order.total_cents) {
      return reply.code(422).send({ error: "amount exceeds order total" });
    }

    // (2) احجز المفتاح الأول: الـ UNIQUE هو اللي بيمنع الـ refund المزدوج، مش if
    const claim = await db.query<{ id: number }>(
      `INSERT INTO refunds (tenant_id, idempotency_key, order_id, amount_cents, reason, status)
       VALUES ($1, $2, $3, $4, $5, 'pending')
       ON CONFLICT (tenant_id, idempotency_key) DO NOTHING
       RETURNING id`,
      [tenantId, key, order.id, body.amountCents, body.reason],
    );
    const refund = claim.rows[0];
    if (!refund) return reply.send({ ok: true, replayed: true });

    // (3) الـ error بيوصل للي طلب: 502 مش 200 مع log
    try {
      await gateway.refund(order.payment_ref, body.amountCents);
    } catch (err) {
      await db.query("UPDATE refunds SET status = 'failed' WHERE id = $1", [refund.id]);
      req.log.error({ err, orderId: order.id }, "refund failed at gateway");
      return reply.code(502).send({ error: "refund failed" });
    }

    // (4) بنعملها في statement واحد بدل loop: عدد الـ queries ثابت مهما كان عدد السطور
    await db.query("UPDATE refunds SET status = 'done' WHERE id = $1", [refund.id]);
    await db.query("UPDATE orders SET status = 'refunded' WHERE id = $1 AND tenant_id = $2", [order.id, tenantId]);
    await db.query(
      `UPDATE products p SET stock = p.stock + l.qty
       FROM order_lines l WHERE l.order_id = $1 AND l.sku = p.sku`,
      [order.id],
    );
    return reply.send({ ok: true, refundId: refund.id });
  });
}

// استخدمه كده: نادي ⁦tick()⁩ مرة لكل وحدة شغل وضاعف الحجم
// لو الحل ⁦O(n)⁩ هتشوف x2.00 كل مرة، و ⁦O(n^2)⁩ هتشوف x4.00، و ⁦O(n log n)⁩ فوق x2 شوية
export function doubling(name: string, run: (n: number, tick: () => void) => void, sizes: number[]): void {
  let prev: { ops: number; ms: number } | null = null;
  for (const n of sizes) {
    let ops = 0;
    const started = performance.now();
    run(n, () => { ops++; });
    const ms = performance.now() - started;
    const opsRatio = prev ? 'x' + (ops / prev.ops).toFixed(2) : '';
    const msRatio = prev ? 'x' + (ms / Math.max(prev.ms, 0.01)).toFixed(2) : '';
    console.log(`${name.padEnd(10)} n=${String(n).padStart(8)} ops=${String(ops).padStart(12)} ${opsRatio.padEnd(6)} ms≈${ms.toFixed(1)} ${msRatio}`);
    prev = { ops, ms };
  }
}

The clock for a 30-minute review. The first 3 minutes go to intent: read the description and the tests, and say what this change is for. Then design and logic, then security and edge cases, and naming last. A workable split is 3, 10, 12, 3, plus 2 minutes to write the summary. Naming goes last because it's cheap to fix and it's where nervous reviewers hide. If you spot the injection at minute one, note it and keep going: the order decides what you spend time on, not what you are allowed to notice.

Model answer for the second PR. Blocking: the query filters on tenant_id, status and due_on, sorts by due_on, and the table has no index except the primary key. Don't assert it, show the plan: EXPLAIN (ANALYZE, BUFFERS) runs a Seq Scan over the whole table. With a composite index on (tenant_id, status, due_on), equality columns first and the range and sort column last, it becomes an Index Scan. Buffers go from 1,471 to 24, measured below, and the sort disappears. On a large live table the migration should use CREATE INDEX CONCURRENTLY, so writes aren't blocked. Every index also costs write time, which you can reduce with a partial index on WHERE status = 'open' when open invoices are the minority; details are in section 07, B-tree indexes.

export type Size = "small" | "medium" | "large";

const RANK: Record<Size, number> = { small: 0, medium: 1, large: 2 };

export const fits = (item: Size, slot: Size): boolean => RANK[item] <= RANK[slot];

export type Item = { sku: string; size: Size; tenantId: string };

export class Slot {
  readonly id: string;
  readonly size: Size;
  readonly distanceToDock: number;
  #item: Item | null = null;

  constructor(id: string, size: Size, distanceToDock: number) {
    this.id = id;
    this.size = size;
    this.distanceToDock = distanceToDock;
  }

  get isFree(): boolean {
    return this.#item === null;
  }

  place(item: Item): void {
    if (!this.isFree) throw new Error(`slot ${this.id} is occupied`);
    if (!fits(item.size, this.size)) throw new Error(`${item.size} does not fit in ${this.size}`);
    this.#item = item;
  }

  vacate(): Item {
    if (this.#item === null) throw new Error(`slot ${this.id} is already empty`);
    const item = this.#item;
    this.#item = null;
    return item;
  }
}

// ده السطر اللي بيخلي القرار يتغيّر من غير ما الـ allocator يتغيّر
export interface AllocationStrategy {
  pick(item: Item, candidates: readonly Slot[]): Slot | undefined;
}

export class TightestFit implements AllocationStrategy {
  pick(item: Item, candidates: readonly Slot[]): Slot | undefined {
    return [...candidates].sort(
      (a, b) => RANK[a.size] - RANK[b.size] || a.distanceToDock - b.distanceToDock,
    )[0];
  }
}

export class NearestDock implements AllocationStrategy {
  pick(_item: Item, candidates: readonly Slot[]): Slot | undefined {
    return [...candidates].sort((a, b) => a.distanceToDock - b.distanceToDock)[0];
  }
}

export class Warehouse {
  #slots: readonly Slot[];
  #strategy: AllocationStrategy;
  #tickets = new Map<string, Slot>();
  #nextTicket = 1;

  constructor(slots: readonly Slot[], strategy: AllocationStrategy) {
    this.#slots = slots;
    this.#strategy = strategy;
  }

  // بيرجع رقم تذكرة، أو null لو مفيش مكان: ده مسار طبيعي مش error
  store(item: Item): string | null {
    const candidates = this.#slots.filter((s) => s.isFree && fits(item.size, s.size));
    const slot = this.#strategy.pick(item, candidates);
    if (slot === undefined) return null;
    slot.place(item);
    const ticket = `T${this.#nextTicket++}`;
    this.#tickets.set(ticket, slot);
    return ticket;
  }

  release(ticket: string): Item {
    const slot = this.#tickets.get(ticket);
    if (slot === undefined) throw new Error(`unknown ticket ${ticket}`);
    this.#tickets.delete(ticket);
    return slot.vacate();
  }

  get freeSlots(): number {
    return this.#slots.filter((s) => s.isFree).length;
  }
}
import type { AllocationStrategy, Item, Slot } from "./warehouse.ts";

// الإضافة: strategy بتلف strategy تانية (decorator)، والـ Warehouse متلمسش سطر فيه
export class ReservedSlots implements AllocationStrategy {
  #inner: AllocationStrategy;
  #owners: ReadonlyMap<string, string>;

  constructor(inner: AllocationStrategy, owners: ReadonlyMap<string, string>) {
    this.#inner = inner;
    this.#owners = owners;
  }

  pick(item: Item, candidates: readonly Slot[]): Slot | undefined {
    const allowed = candidates.filter((slot) => {
      const owner = this.#owners.get(slot.id);
      return owner === undefined || owner === item.tenantId;
    });
    return this.#inner.pick(item, allowed);
  }
}

import { doubling } from "../doubling.ts";

type Size = "small" | "medium" | "large";
type Slot = { id: number; size: Size; free: boolean };
const ORDER: readonly Size[] = ["small", "medium", "large"];

const makeSlots = (n: number): Slot[] =>
  Array.from({ length: n }, (_, i) => ({ id: i, size: ORDER[i % 3]!, free: true }));

// (1) الـ brute force: كل store بيمشي على كل الأماكن، يبقى ⁦O(S)⁩ للواحد و ⁦O(n·S)⁩ للكل
function fillByScanning(n: number, tick: () => void): void {
  const slots = makeSlots(n);
  for (let item = 0; item < n; item++) {
    let chosen: Slot | undefined;
    for (const slot of slots) {
      tick();
      if (slot.free && chosen === undefined) chosen = slot;
    }
    if (chosen) chosen.free = false;
  }
}

// (2) الحل: free list لكل size، يبقى ⁦O(1)⁩ للواحد و ⁦O(n)⁩ للكل
function fillByFreeLists(n: number, tick: () => void): void {
  const freeBySize = new Map<Size, Slot[]>(ORDER.map((size) => [size, []]));
  for (const slot of makeSlots(n)) freeBySize.get(slot.size)!.push(slot);
  for (let item = 0; item < n; item++) {
    for (const size of ORDER) {
      tick();
      const slot = freeBySize.get(size)!.pop();
      if (slot) {
        slot.free = false;
        break;
      }
    }
  }
}

Model solution, same six steps. Clarify: one lift, floors 0 to 19, unlimited capacity to keep the model small, and every ride waits at its floor until the lift comes. Entities: rides, stops, and the three pieces that must stay separate. Scheduler decides the next stop. A pure transition function owns the lift's states. LiftController connects them and owns the clock.

The state machine. Three states, idle, moving and doors_open, and a transition table where anything unlisted throws. That's what makes "doors open while moving" impossible rather than merely unlikely. Compare a pile of booleans like isMoving and isDoorOpen, where the illegal combination is one forgotten line away. See section 08, State machines.

Brute force first. FCFS serves rides strictly in order, and it's the honest baseline: total travel is O(rides). SCAN keeps going one way, serves every stop on the path, and reverses at the end. With all rides already waiting and unlimited capacity, SCAN's travel is bounded by a couple of sweeps, so it's O(floors) and doesn't grow with n. The output below confirms it.

Trade-off. Measured on 60 seeded rides, FCFS travelled 881 floors and SCAN 119, and SCAN also waited far less. FCFS is the simplest and easiest to reason about, while SCAN is what real lifts do because it bounds both travel and waiting. A greedy "nearest stop first" looks attractive but can starve a far floor, which is the follow-up question in the callout below.

export type Direction = "up" | "down";
export type Ride = { id: string; at: number; from: number; to: number };
export type Stop = { rideId: string; floor: number; kind: "pickup" | "dropoff" };

export interface Scheduler {
  // بيقول الـ stop الجاي إيه، والـ controller هو اللي بيتحرك
  next(floor: number, direction: Direction, stops: readonly Stop[]): Stop | undefined;
  // بيقول مين هيتخدم لما الباب يفتح عند الـ stop ده

}

export class Fcfs implements Scheduler {

// ده الـ elevator algorithm: بيكمّل في نفس الاتجاه لآخره وبعدين يلف
export class Scan implements Scheduler {
  // الدور اللي لسه خدمناه: اللي يوصل بعدها يستنى الـ sweep الجاي، وإلا الدور ده هيمسكنا
  #justServed: number | null = null;

  next(floor: number, direction: Direction, stops: readonly Stop[]): Stop | undefined {
    if (stops.length === 0) return undefined;
    const here = stops.find((s) => s.floor === floor);
    if (here !== undefined && this.#justServed !== floor) return here;
    const ahead = (dir: Direction) =>
      stops
        .filter((s) => (dir === "up" ? s.floor > floor : s.floor < floor))
        .sort((a, b) => Math.abs(a.floor - floor) - Math.abs(b.floor - floor))[0];
    return ahead(direction) ?? ahead(direction === "up" ? "down" : "up") ?? here;
  }

  serveWith(chosen: Stop, stops: readonly Stop[]): Stop[] {
    this.#justServed = chosen.floor;
    return stops.filter((s) => s.floor === chosen.floor);
  }
}

export type LiftState =
  | { kind: "idle" }
  | { kind: "moving"; direction: Direction; target: number }
  | { kind: "doors_open"; ticksLeft: number };

export type LiftEvent =
  | { type: "depart"; direction: Direction; target: number }
  | { type: "arrive" }
  | { type: "open"; ticks: number }
  | { type: "close" };

// الـ state machine: كل انتقال غير مسموح بيرمي error بدل ما يعدي بهدوء
export function transition(state: LiftState, event: LiftEvent): LiftState {
  switch (state.kind) {
    case "idle":
      if (event.type === "depart") return { kind: "moving", direction: event.direction, target: event.target };
      if (event.type === "open") return { kind: "doors_open", ticksLeft: event.ticks };
      break;
    case "moving":
      if (event.type === "arrive") return { kind: "idle" };
      break;
    case "doors_open":
      if (event.type === "close") return { kind: "idle" };
      break;
  }
  throw new Error(`illegal transition: ${state.kind} + ${event.type}`);
}

export type Metrics = { floorsTravelled: number; ticks: number; avgWait: number; maxWait: number };

const DOOR_TICKS = 2;

export class LiftController {
  #scheduler: Scheduler;
  #floor = 0;
  #direction: Direction = "up";
  #state: LiftState = { kind: "idle" };

  constructor(scheduler: Scheduler) {
    this.#scheduler = scheduler;
  }

  // المحاكاة deterministic: tick واحدة لكل دور، ومفيش وقت حقيقي
  run(rides: readonly Ride[], onFloorMoved: () => void = () => {}): Metrics {
    for (const ride of rides) {
      if (ride.from === ride.to) throw new Error(`ride ${ride.id} goes nowhere`);
    }
    const waiting = [...rides].sort((a, b) => a.at - b.at);
    // الدور بيفضل محفوظ: الـ dropoff بياخد مكان الـ pickup بتاع نفس الـ ride
    const open: Stop[] = [];
    const pickedAt = new Map<string, number>();
    let floorsTravelled = 0;
    let tick = 0;

    const add = (ride: Ride) => open.push({ rideId: ride.id, floor: ride.from, kind: "pickup" });
    const byId = new Map(rides.map((r) => [r.id, r]));

    while (waiting.length > 0 || open.length > 0 || this.#state.kind !== "idle") {
      while (waiting.length > 0 && waiting[0]!.at <= tick) add(waiting.shift()!);

      if (this.#state.kind === "doors_open") {
        const left = this.#state.ticksLeft - 1;
        this.#state = left > 0 ? { kind: "doors_open", ticksLeft: left } : transition(this.#state, { type: "close" });
        tick++;
        continue;
      }

      if (this.#state.kind === "moving") {
        const { target, direction } = this.#state;
        this.#floor += direction === "up" ? 1 : -1;
        floorsTravelled++;
        onFloorMoved();
        tick++;
        if (this.#floor === target) this.#state = transition(this.#state, { type: "arrive" });
        continue;
      }

      // هنا الحالة idle: يا إما نفتح الباب في الدور ده، يا نطلب الـ stop الجاي
      const chosen = this.#scheduler.next(this.#floor, this.#direction, open);
      if (chosen === undefined) {
        tick++;
        continue;
      }
      if (chosen.floor === this.#floor) {
        for (const stop of this.#scheduler.serveWith(chosen, open)) {
          const at = open.indexOf(stop);
          if (stop.kind === "dropoff") {
            open.splice(at, 1);
            continue;
          }
          pickedAt.set(stop.rideId, tick);
          open[at] = { rideId: stop.rideId, floor: byId.get(stop.rideId)!.to, kind: "dropoff" };
        }
        this.#state = transition(this.#state, { type: "open", ticks: DOOR_TICKS });
        continue;
      }
      this.#direction = chosen.floor > this.#floor ? "up" : "down";
      this.#state = transition(this.#state, { type: "depart", direction: this.#direction, target: chosen.floor });
    }

    const waits = rides.map((r) => pickedAt.get(r.id)! - r.at);
    return {
      floorsTravelled,
      ticks: tick,
      avgWait: Math.round(waits.reduce((a, b) => a + b, 0) / Math.max(waits.length, 1)),
      maxWait: Math.max(0, ...waits),
    };
  }
}

// مولّد أرقام بـ seed عشان نفس الـ rides تتولد كل مرة (mulberry32)
export function seededRandom(seed: number): () => number {
  let a = seed;
  return () => {
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

export function makeRides(count: number, floors: number, seed: number, gap: number): Ride[] {
  const random = seededRandom(seed);
  const rides: Ride[] = [];
  for (let i = 0; i < count; i++) {
    const from = Math.floor(random() * floors);
    let to = Math.floor(random() * floors);
    if (to === from) to = (to + 1) % floors;
    rides.push({ id: `r${i}`, at: i * gap, from, to });
  }
  return rides;
}

English

Model solution. Clarify, then entities: RentalDesk runs the flow. ReservationStore is an interface, so the logic doesn't know about SQL. FinePolicy is a small interface with DailyFine and a CappedFine wrapper, which is composition rather than a LateLaptopRental extends LaptopRental tree. Clock is injected so tests move time without waiting. Fines and due dates are integers and Dates; the same money rule applies as in Part 1.

The race, and the guard. The naive version checks for an active reservation and then inserts. There's an await between those two steps, so two borrowers both find nothing and both insert. The guard is a partial unique index, one active reservation per tenant and copy, with INSERT ... ON CONFLICT DO NOTHING RETURNING so the loser gets zero rows instead of an error. Run on PGlite, naive double-booked 100 of 100 rounds. On a real Postgres 18 with a pool of 8 connections, it double-booked 98 of 100. With the guard, both were 0 of 100. PGlite has one connection, so its queries take turns, while the JavaScript still interleaves between them; the Postgres run is the one with real concurrent connections.

When a pattern is too much. One fine rule that never changes should be a function, not a Strategy. Reach for FinePolicy because the interviewer said "daily fine with a cap, and other tenants may want different rules". Say that reason out loud, and see section 08, Pattern judgement. A pattern with no second implementation and no stated reason is just more code to review.

Complexity, brute force first. Answering "is this copy free?" by scanning every reservation is O(R) per question and O(n·R) across a day of questions. The index makes it O(1) per question. In-memory, that's a Map; in Postgres, it's the unique index that the guard already gave you. The counts below confirm x4.00 against x2.00.

export interface FinePolicy {
  fineMinor(daysLate: number): number;

export class DailyFine implements FinePolicy {
  #ratePerDayMinor: number;
  constructor(ratePerDayMinor: number) {
    this.#ratePerDayMinor = ratePerDayMinor;
  }
  fineMinor(daysLate: number): number {
    return daysLate * this.#ratePerDayMinor;
  }
}

export class CappedFine implements FinePolicy {
  #inner: FinePolicy;
  #capMinor: number;
  constructor(inner: FinePolicy, capMinor: number) {
    this.#inner = inner;
    this.#capMinor = capMinor;
  }
  fineMinor(daysLate: number): number {
    return Math.min(this.#inner.fineMinor(daysLate), this.#capMinor);
  }
}

  tryInsert(input: Omit<Reservation, "id">): Promise<Reservation | null>;

export class AlreadyReserved extends Error {}

export class RentalDesk {
  #store: ReservationStore;
  #fines: FinePolicy;
  #clock: Clock;
  #loanDays: number;

  constructor(store: ReservationStore, fines: FinePolicy, clock: Clock, loanDays = 7) {
    this.#store = store;
    this.#fines = fines;
    this.#clock = clock;
    this.#loanDays = loanDays;
  }

  #dueAt(): Date {
    return new Date(this.#clock().getTime() + this.#loanDays * DAY_MS);
  }

  // ❌ النسخة الأولى: فحص وبعدين كتابة، وبينهم await
  async reserveNaive(tenantId: string, copyId: string, borrowerId: string): Promise<Reservation> {
    if (await this.#store.findActive(tenantId, copyId)) throw new AlreadyReserved(copyId);
    return this.#store.insertUnchecked({ tenantId, copyId, borrowerId, dueAt: this.#dueAt() });
  }

  // ✅ الحل: مفيش فحص قبل الكتابة، الـ DB هي اللي بتحكم
  async reserve(tenantId: string, copyId: string, borrowerId: string): Promise<Reservation> {
    const created = await this.#store.tryInsert({ tenantId, copyId, borrowerId, dueAt: this.#dueAt() });
    if (created === null) throw new AlreadyReserved(copyId);
    return created;
  }

  // بيرجّع الغرامة بالقروش، والتاريخ جاي من الساعة المحقونة
  async giveBack(reservationId: number): Promise<{ fineMinor: number }> {
    const r = await this.#store.get(reservationId);
    if (r === null) throw new Error(`unknown reservation ${reservationId}`);
    const now = this.#clock();
    const closed = await this.#store.markReturned(r.id, now);
    if (!closed) throw new Error(`reservation ${r.id} already returned`);
    const daysLate = Math.max(0, Math.ceil((now.getTime() - r.dueAt.getTime()) / DAY_MS));
    return { fineMinor: this.#fines.fineMinor(daysLate) };
  }
}

export const RESERVATIONS_SCHEMA = `
  CREATE TABLE reservations (
    id          serial PRIMARY KEY,
    tenant_id   text NOT NULL,
    copy_id     text NOT NULL,
    borrower_id text NOT NULL,
    due_at      timestamptz NOT NULL,
    returned_at timestamptz
  );
`;
export const GUARD_INDEX = `
  CREATE UNIQUE INDEX one_active_reservation
    ON reservations (tenant_id, copy_id) WHERE returned_at IS NULL;
`;

type Row = { id: number; tenant_id: string; copy_id: string; borrower_id: string; due_at: Date };

const fromRow = (r: Row): Reservation => ({
  id: r.id,
  tenantId: r.tenant_id,
  copyId: r.copy_id,
  borrowerId: r.borrower_id,
  dueAt: new Date(r.due_at),
});

export class SqlReservationStore implements ReservationStore {
  #db: Queryable;
  constructor(db: Queryable) {
    this.#db = db;
  }

  async findActive(tenantId: string, copyId: string): Promise<Reservation | null> {
    const res = await this.#db.query<Row>(
      "SELECT * FROM reservations WHERE tenant_id = $1 AND copy_id = $2 AND returned_at IS NULL",
      [tenantId, copyId],
    );
    return res.rows[0] ? fromRow(res.rows[0]) : null;
  }

  async insertUnchecked(i: Omit<Reservation, "id">): Promise<Reservation> {
    const res = await this.#db.query<Row>(
      `INSERT INTO reservations (tenant_id, copy_id, borrower_id, due_at)
       VALUES ($1, $2, $3, $4) RETURNING *`,
      [i.tenantId, i.copyId, i.borrowerId, i.dueAt],
    );
    return fromRow(res.rows[0]!);
  }

  // (2) الحل في statement واحد: اللي يكسب الـ index ياخد الـ row، واللي يخسر ياخد 0 rows
  async tryInsert(i: Omit<Reservation, "id">): Promise<Reservation | null> {
    const res = await this.#db.query<Row>(
      `INSERT INTO reservations (tenant_id, copy_id, borrower_id, due_at)
       VALUES ($1, $2, $3, $4)
       ON CONFLICT (tenant_id, copy_id) WHERE returned_at IS NULL DO NOTHING
       RETURNING *`,
      [i.tenantId, i.copyId, i.borrowerId, i.dueAt],
    );
    return res.rows[0] ? fromRow(res.rows[0]) : null;
  }

  async get(id: number): Promise<Reservation | null> {
    const res = await this.#db.query<Row>("SELECT * FROM reservations WHERE id = $1", [id]);
    return res.rows[0] ? fromRow(res.rows[0]) : null;
  }

  async markReturned(id: number, returnedAt: Date): Promise<boolean> {
    const res = await this.#db.query<Row>(
      "UPDATE reservations SET returned_at = $2 WHERE id = $1 AND returned_at IS NULL RETURNING *",
      [id, returnedAt],
    );
    return res.rows.length === 1;
  }
}

type Reservation = { copyId: string; active: boolean };

const makeReservations = (n: number): Reservation[] =>
  Array.from({ length: n }, (_, i) => ({ copyId: `copy-${i}`, active: i % 2 === 0 }));

// (1) الـ brute force: السؤال "النسخة دي متاحة؟" بيعدّي على كل الحجوزات، ⁦O(R)⁩ للسؤال الواحد
function askByScanning(n: number, tick: () => void): void {
  const all = makeReservations(n);
  for (let i = 0; i < n; i++) {
    const wanted = `copy-${i}`;
    let taken = false;
    for (const r of all) {
      tick();
      if (r.copyId === wanted && r.active) taken = true;
    }
    if (taken === (i % 2 !== 0)) throw new Error("wrong answer");
  }
}

// (2) الحل: Map بالـ copy، يبقى ⁦O(1)⁩ للسؤال. في SQL ده هو الـ index اللي فوق
function askByIndex(n: number, tick: () => void): void {
  const byCopy = new Map(makeReservations(n).map((r) => [r.copyId, r]));
  for (let i = 0; i < n; i++) {
    tick();
    const taken = byCopy.get(`copy-${i}`)?.active ?? false;
    if (taken === (i % 2 !== 0)) throw new Error("wrong answer");
  }
}

الغلطة إنك تصلّح الـ race بفحص قبل الـ insert أو lock في process واحد. الاتنين مش بيصمدوا قدام Node instance تانية. الـ constraint في الـ DB بيصمد، والكود بعدها كل اللي عليه إنه يقرا نتيجة الـ insert.
Interview question "How do you stop a double booking without a distributed lock?" → Let the database decide with a unique constraint and ON CONFLICT DO NOTHING. It's one atomic statement, it works across every Node instance, and the loser gets zero rows to report as "already taken". Reach for a distributed lock only when no constraint can express the rule.
"إزاي تمنع الحجز المزدوج من غير distributed lock؟" ← سيب الـ DB تقرر بـ unique constraint وON CONFLICT DO NOTHING. ده statement واحد atomic، شغال على كل الـ Node instances، واللي يخسر بياخد صفر rows فتقوله "النسخة اتحجزت". الـ distributed lock تلجأ له بس لما مفيش constraint يقدر يعبّر عن القاعدة.

import { LiftController, Scan, type Direction, type Ride, type Scheduler, type Stop } from "./lift.ts";

// ❌ أقرب stop الأول: بيقلل الحركة، بس ممكن يسيب دور بعيد مستني طول ما فيه زحمة
export class NearestFirst implements Scheduler {
  next(floor: number, _direction: Direction, stops: readonly Stop[]): Stop | undefined {
    return [...stops].sort((a, b) => Math.abs(a.floor - floor) - Math.abs(b.floor - floor))[0];
  }
  serveWith(chosen: Stop, stops: readonly Stop[]): Stop[] {
    return stops.filter((s) => s.floor === chosen.floor);
  }
}

// (1) مشوار واحد في الدور 19، وزحمة مستمرة بين الدورين 0 و 2 لمدة 400 tick

for (let i = 0; i < 400; i++) rides.push({ id: `near${i}`, at: i, from: i % 2, to: 2 });

Cheat sheet · الخلاصة

  • Debugging loop: reproduce (failing test) → isolate (shrink, one hypothesis aloud) → fix the cause → verify (same test + one neighbouring case).
  • Money is integer minor units, with one exact division at the end. toFixed rounds the same wrong double.
  • Deep-freeze a fixture and the first mutation throws with the exact line.
  • An await between read and write means a lost update. Make check-and-write one atomic step.
  • A MaxListenersExceededWarning prints once, at the 11th listener. --trace-warnings finds it; never silence it.
  • Review order: 3 minutes intent → design and logic → security and edge cases → naming last.
  • Label every review comment blocking or nit and say what breaks and for whom. Prove the top issue.
  • Parameterised SQL fixes injection; a tenant filter is a separate fix.
  • LLD steps: clarify 5, entities 5, interfaces 5, key flow 5, code 20, extension 5.
  • A new requirement should arrive as a new class, not a new if inside the allocator. Illegal states shouldn't be expressible.
  • Two callers, one resource: a database constraint with ON CONFLICT DO NOTHING, never a check before the insert.
  • A pattern needs a reason you can say: one fixed rule is a function.
بالعامية
  • دايرة الـ debugging: reproduce (test فاشل) ثم isolate (صغّر، و hypothesis واحد بصوت عالي) ثم صلّح السبب ثم verify (نفس الـ test وحالة جارة).
  • الفلوس قروش integers وقسمة واحدة مظبوطة في الآخر. toFixed بيقرّب نفس الـ double الغلط.
  • اعمل deep freeze للـ fixture وأول تعديل هيرمي error بالسطر بالظبط.
  • await بين القراءة والكتابة يبقى lost update. خلي الفحص والكتابة خطوة atomic واحدة.
  • MaxListenersExceededWarning بيتطبع مرة واحدة عند الـ listener رقم 11. --trace-warnings بيوصّلك للمكان، وعمرك ما تسكّته.
  • ترتيب الـ review: 3 دقايق intent ثم design و logic ثم security و edge cases ثم الـ naming في الآخر.
  • كل comment يتعلّم blocking أو nit وقول إيه اللي بيكسر ولمين. وأثبت أكبر مشكلة.
  • الـ SQL بالـ parameters بيحل الـ injection، والـ tenant filter تصليح منفصل.
